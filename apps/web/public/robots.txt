# HelpDesk Lite
# La aplicación es una SPA privada: no hay nada que indexar detrás del login,
# y la API no es contenido. Staging debería además quedar tras una Access List
# de Cloudflare (no depende de este fichero).
User-agent: *
Allow: /$
# La página de estado sí es contenido público: es la que se mira cuando el
# servicio falla, y no hay nada privado en ella.
Allow: /status
Disallow: /api/
Disallow: /socket.io/

# La documentación de la API no es contenido público: además va con
# contraseña y con X-Robots-Tag, esto es sólo la primera señal.
Disallow: /api/v1/docs
Disallow: /api/v1/openapi.json
